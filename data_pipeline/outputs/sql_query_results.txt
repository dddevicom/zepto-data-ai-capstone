QUERY 5: Books from selected categories using IN
--------------------------------------------------------------------------------
SELECT
    b.title,
    b.price_gbp,
    b.rating,
    c.category_name
FROM books AS b
JOIN categories AS c
    ON b.category_id = c.category_id
WHERE c.category_name IN (
    SELECT category_name
    FROM categories
    ORDER BY category_name
    LIMIT 3
)
ORDER BY c.category_name, b.title

QUERY 6: Category join summary
--------------------------------------------------------------------------------
SELECT
    c.category_name,
    COUNT(b.book_id) AS book_count,
    ROUND(AVG(b.price_gbp), 2) AS average_price_gbp
FROM categories AS c
JOIN books AS b
    ON b.category_id = c.category_id
GROUP BY c.category_id, c.category_name
ORDER BY book_count DESC, c.category_name

     category_name  book_count  average_price_gbp
           Mystery          32              31.72
Historical Fiction          26              33.64
            Travel          11              39.79
